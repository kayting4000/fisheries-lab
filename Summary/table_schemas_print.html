Star Schema Overview

dim_date
  • PK date_key INT
  • full_date DATE
  • day SMALLINT
  • month SMALLINT
  • month_name VARCHAR
  • quarter SMALLINT
  • year SMALLINT
  • week_of_year SMALLINT
  • day_of_week SMALLINT
  • day_name VARCHAR
  • is_weekend BOOL
fact_landing
  • PK landing_key BIGSERIAL
  • FK date_key INT
  • FK port_key INT
  • FK species_key INT
  • FK vessel_key INT
  • quantity_kg NUMERIC
  • price_per_kg NUMERIC
  • total_value NUMERIC (generated)
  • trip_days SMALLINT
dim_port
  • PK port_key SERIAL
  • port_code VARCHAR(10)
  • port_name VARCHAR(100)
  • municipality VARCHAR
  • province VARCHAR
  • region VARCHAR
  • cooperative VARCHAR
dim_species
  • PK species_key SERIAL
  • species_code VARCHAR(10)
  • common_name VARCHAR
  • scientific_name VARCHAR
  • family VARCHAR
  • category VARCHAR
dim_vessel
  • PK vessel_key SERIAL
  • vessel_code VARCHAR(20)
  • vessel_name VARCHAR
  • vessel_type VARCHAR
  • gross_tonnage NUMERIC
  • owner_name VARCHAR
  • cooperative VARCHAR
  • effective_date DATE
  • expiry_date DATE
  • is_current BOOL (SCD2)
